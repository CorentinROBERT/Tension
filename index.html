  <h2>Nouvelle mesure</h2>

  <input type="number" id="sys" placeholder="Systole (ex: 120)">

  <input type="number" id="dia" placeholder="Diastole (ex: 80)">

  <input type="number" id="pulse" placeholder="Pouls (ex: 65)">

  <button onclick="addEntry()">Enregistrer</button>

</div>


<div class="card">

  <canvas id="tensionChart"></canvas>

</div>


<script>

  let data = JSON.parse(localStorage.getItem('bp_data')) || [];

  let chart;


  function initChart() {

    const ctx = document.getElementById('tensionChart').getContext('2d');

    chart = new Chart(ctx, {

      type: 'line',

      data: {

        labels: data.map(e => e.date),

        datasets: [

          { label: 'Systole', data: data.map(e => e.sys), borderColor: 'red', backgroundColor: 'red', fill: false },

          { label: 'Diastole', data: data.map(e => e.dia), borderColor: 'blue', backgroundColor: 'blue', fill: false },

          { label: 'Pouls', data: data.map(e => e.pulse), borderColor: 'green', backgroundColor: 'green', fill: false }

        ]

      }

    });

  }


  function addEntry() {

    const sys = document.getElementById('sys').value;

    const dia = document.getElementById('dia').value;

    const pulse = document.getElementById('pulse').value;

    if (!sys || !dia || !pulse) return;


    const date = new Date().toLocaleDateString('fr-FR', { day: '2-digit', month: '2-digit' });

    data.push({ date, sys, dia, pulse });

    localStorage.setItem('bp_data', JSON.stringify(data));

    

    chart.destroy();

    initChart();

  }


  initChart();

</script>
